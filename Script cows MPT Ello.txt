procedure PITCH
# adapated from David Reby, Frederique Sebe
	
	select current_sound
	sound_duration = Get total duration
	Edit

	editor Sound 'name$'
   		Spectrogram settings... 0 sview_range_max swindow_length 60 
	endeditor

	To Pitch (cc)... tiste minF0 maxCandF0 no silThreF0 voicThreF0 octCostF0 octJumpCostF0 voicUnvoicCostF0 maxF0
	Rename... Pitch
	Edit
	pause Inspect the sound
	Smooth... 10
	
	#pause Inspect the sound
	Rename... Smoothed
	Interpolate
	Rename... Interpolated
	
	f0_mean = Get mean... 0 0 Hertz
	f0_max = Get maximum... 0 0 Hertz Parabolic
	f0_min = Get minimum... 0 0 Hertz Parabolic
	f0_range = f0_max - f0_min
	
	Down to PitchTier
	select Pitch Interpolated
	Down to PitchTier
	Down to TableOfReal... Hertz
	Rename... voiced
	select TableOfReal voiced
	nrow = Get number of rows
	Remove

	print ANALYSIS SUMMARY'newline$'
	print File name: 'sound$''newline$'
	print Call type = 'call_type$''newline$'
	printline
	print Mean F0 = 'f0_mean:3' Hz'newline$'
	print Max F0 = 'f0_max:3' Hz'newline$'
	print Min F0 = 'f0_min:3' Hz'newline$'
	print Range F0 = 'f0_range:3'Hz'newline$'
		
	fileappend "'pf_spreadsheet$'" 'newline$''sound$','call_type$','f0_mean:3','f0_max:3','f0_min:3','f0_range:3',
	#File name,Call type,Mean F0,Max F0,Min F0,Range F0,
endproc

procedure AMPLITUDE
# adapated from David Reby, Ben Charlton, Frederique Sebe
		
	select current_sound
	sound_duration = Get total duration
	To Intensity... minF0 tiste yes
	Rename...  int_son
		tmaxAMP = Get time of maximum... 0 0 Parabolic
		tmaxAMP2 = (tmaxAMP / sound_duration)*100
	
	Down to IntensityTier

	#clearinfo
	nframe = Get number of points
	infl_asc = 0
	infl_desc = 0
	rowin = 0
	variationtot = 0
	variation = 0

	Create TableOfReal... formul 1 2
	Set row label (label)... "" rowin
	Set column label (index)... 1 frame
	Set column label (index)... 2 in
	
	select Intensity int_son
	for current_frame from 1 to nframe-1
		intensity_frame = Get value in frame... current_frame 
		intensity_frame_before = Get value in frame... current_frame-1 
		intensity_frame_after = Get value in frame... current_frame+1 

		if intensity_frame <> undefined and intensity_frame_before <> undefined and intensity_frame_after <> undefined
			if intensity_frame_after >  intensity_frame and intensity_frame <= intensity_frame_before
				infl_asc +=1
			elsif intensity_frame_after < intensity_frame and intensity_frame >= intensity_frame_before
				infl_desc +=1
			endif
			variationtot += abs(intensity_frame_after - intensity_frame)
			variation += (intensity_frame_after - intensity_frame)
			rowin += 1
			select TableOfReal formul
			Insert row (index)... rowin
			Set value... rowin 2 rowin 
			Set value... rowin 2 intensity_frame
			select Intensity int_son

		endif	
	endfor
	
	variation_abs = abs(variation)
	sum_infl = (infl_asc + infl_desc)
	inflex = sum_infl/sound_duration
	sumvar = variation_abs/sound_duration
	#cumulative variation in the In contour divided by call duration
	#mean variation per second (AM var):
	amvar = variationtot/sound_duration
	#number of complete cycles of In modulation per second (AM rate):
	amrate = ((sum_infl)/2)/sound_duration
	#mean peak-to-peak variation of each In modulation per second (AM extent):
	amextent = (variationtot/((sum_infl)/2))
	
	print sound duration = 'sound_duration:3' sec.'newline$'
	print AM var  = 'amvar:3' dB'newline$'
	print AM rate = 'amrate:3' cps'newline$'
	print AM extent = 'amextent:3' dB'newline$'
	fileappend "'pf_spreadsheet$'" 'sound_duration:3','amvar:3','amrate:3','amextent:3',
	#sound duration,AM var,AM rate,AM extent,

endproc

# Wiener entropy, by Gabriel J.L. Beckers
# developed in Praat version 4.2.06
# 2004-07-03

# Wiener entropy is a measure of spectral flatness of a sound.
# It is defined as the ratio of a power spectrum's geometric mean and
# its arithmetic mean. It is scaled to a log scale, which ranges from
# 0 (power spectrum is flat) to minus infinity (power spectrum is 
# infinitely narrow).

####################

	select current_sound

	frame_duration = 0.01
	time_stepWE = 0.004
	start_frequency = 50
	end_frequency = q75 
	#this end frequency allows for normalizing data as wiener entropy intrinsically increases with the bandwidth of the sound
	
	sampling_period = Get sampling period
	duration = Get total duration
	start_time = Get starting time
	number_of_steps = floor((duration-frame_duration)/time_stepWE) + 1

	sum_wiener_entropy = 0
	for frame_no from 1 to number_of_steps
		Extract part... start_time start_time+frame_duration Gaussian1 1 yes
		sound_frame_id = selected("Sound")
		#printline 'start_time'
		start_time = start_time + time_stepWE

		To Spectrum (dft)
		spectrum_id = selected("Spectrum")
		Rename... Spectrum
		number_of_bins = Get number of bins
		#print nb of bins = 'number_of_bins'

		highest_frequency = Get highest frequency
		if highest_frequency < end_frequency
			end_frequency = highest_frequency
		endif
		start_bin = Get bin number from frequency... 'start_frequency'
		end_bin = Get bin number from frequency... 'end_frequency'
		start_bin = round(start_bin)
		end_bin = round(end_bin)
		actual_start_frequency = Get frequency from bin number... 'start_bin'
		actual_end_frequency = Get frequency from bin number... 'end_bin'
		number_of_band_bins = end_bin - start_bin + 1

		Create simple Matrix... power_spectrum 1 'number_of_band_bins' 0
		power_spectrum_id = selected("Matrix")
		Formula... ((Spectrum_Spectrum [1,col]/sampling_period)^2 + (Spectrum_Spectrum [2,col]/sampling_period)^2)

		sum_power_spectrum = 0
		for bin from start_bin to end_bin
			sum_power_spectrum += Matrix_power_spectrum[1,bin]
		endfor
		arithmetic_mean = sum_power_spectrum/number_of_band_bins
		#print arith mean = 'arithmetic_mean' 'newline$'
	
		Create simple Matrix... ln_power_spectrum 1 'number_of_band_bins' 0
		ln_power_spectrum_id = selected("Matrix")
		Formula... ln(Matrix_power_spectrum[])

		sum_ln_power_spectrum = 0
		for bin from start_bin to end_bin
			sum_ln_power_spectrum += Matrix_ln_power_spectrum[1,bin]
		endfor
		geometric_mean = exp(sum_ln_power_spectrum/number_of_band_bins)
		#print nb of band bins = 'number_of_band_bins'
		#print geom mean = 'geometric_mean' 'newline$'		

		frame_wiener_entropy = ln(geometric_mean/arithmetic_mean)
		#print 'frame_wiener_entropy' 'newline$'
		#printline wiener entropy frame 'frame_no': 'frame_wiener_entropy'
		sum_wiener_entropy += frame_wiener_entropy

		select 'spectrum_id'
		plus 'power_spectrum_id'	
		plus 'ln_power_spectrum_id'
		plus 'sound_frame_id'
		Remove
		select current_sound
	
	endfor

	wiener_entropy = sum_wiener_entropy/number_of_steps
	print mean wiener entropy = 'wiener_entropy:3' 'newline$' 
	fileappend "'pf_spreadsheet$'" 'wiener_entropy:3',

endproc
